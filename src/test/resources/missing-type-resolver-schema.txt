# The union member type names deliberately match the mock DTO class simple names
# (HumanDTO, DroidDTO) so that a class-name-based type resolution fallback would
# silently succeed here instead of reporting the missing resolver.

type Query {
    unionQuery: CharactersAsUnion @fetcher(name:"character/fetcher")
}

union CharacterUnion @resolver(name:"notRegistered/resolver" source:"CharacterUnion") = HumanDTO | DroidDTO

type HumanDTO {
  id: ID!
  name: String!
  address: String
}

type DroidDTO {
  id: ID!
  name: String!
  primaryFunction: String
}

type CharactersAsUnion {
    characters: [CharacterUnion]
}
